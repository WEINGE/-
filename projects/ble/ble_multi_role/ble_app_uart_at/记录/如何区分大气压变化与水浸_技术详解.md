# 如何区分大气压变化与水浸 - 技术详解

## ? 核心挑战

**问题**: 压力传感器检测到压力上升，如何判断这是由于大气压自然波动还是发生了水浸？

**难度**: 两者都会导致传感器读数增加，单纯依靠压力值无法直接区分

---

## ? 特征对比分析

### 大气压变化的特征

| 维度 | 特征 | 典型值 |
|------|------|--------|
| **变化速度** | 缓慢渐变 | 1-2 hPa/小时 |
| **变化幅度** | 中等 | ±10 hPa (24小时内) |
| **持续性** | 持续性强 | 持续数小时到数天 |
| **周期性** | 有规律 | 随天气系统变化 |
| **温度关联** | 强相关 | 气压升高→温度可能升降 |
| **稳定性** | 相对稳定 | 短期内波动小 |
| **可逆性** | 完全可逆 | 天气系统过境后恢复 |

### 水浸的特征

| 维度 | 特征 | 典型值 |
|------|------|--------|
| **变化速度** | 快速上升 | 5-20 hPa/分钟 |
| **变化幅度** | 大 | >8 hPa (水深>8cm) |
| **持续性** | 持续稳定 | 持续至水退去 |
| **周期性** | 无规律/突发 | 随机事件 |
| **温度关联** | 弱相关 | 水温影响轻微 |
| **稳定性** | 非常稳定 | 水深稳定→压力稳定 |
| **可逆性** | 需排水 | 不会自然恢复 |

## ? 多维度判断方法

### 方法1: 变化速率分析 ?

**原理**: 大气压变化缓慢，水浸压力变化快速

```c
// 计算压力变化速率
float calculate_pressure_change_rate(void) {
    // 获取最近两次采样
    float P_current = latest_pressure;
    float P_previous = previous_pressure;
    uint32_t time_delta = current_time - previous_time;  // 毫秒
    
    // 计算变化率 (hPa/分钟)
    float rate = (P_current - P_previous) / (time_delta / 60000.0f);
    
    return rate;
}

// 判断逻辑
bool is_water_immersion_by_rate(float rate) {
    const float FAST_CHANGE_THRESHOLD = 3.0f;  // hPa/分钟
    
    if (fabs(rate) > FAST_CHANGE_THRESHOLD) {
        // 快速变化 → 可能是水浸
        return true;
    } else {
        // 缓慢变化 → 可能是大气压
        return false;
    }
}
```

**实例**：
```
场景1: 大气压变化
时间: 10:00 → 11:00 (60分钟)
压力: 1013.0 → 1015.0 hPa
变化率: 2.0 / 60 = 0.033 hPa/分钟 ? (缓慢，正常)

场景2: 水浸
时间: 10:00:00 → 10:02:00 (2分钟)
压力: 1013.0 → 1023.0 hPa (10cm水深)
变化率: 10.0 / 2 = 5.0 hPa/分钟 ?? (快速，水浸！)
```

### 方法2: 变化幅度+时间窗口分析 ?

**原理**: 结合变化幅度和时间跨度综合判断

```c
typedef struct {
    float pressure_samples[30];  // 最近30次采样
    uint32_t timestamps[30];
    uint8_t count;
} pressure_history_t;

bool detect_immersion_by_amplitude(pressure_history_t *history, float baseline) {
    if (history->count < 5) return false;  // 样本不足
    
    // 1. 计算最近压力与基准的差值
    float latest_pressure = history->pressure_samples[history->count - 1];
    float pressure_delta = latest_pressure - baseline;
    
    // 2. 检查变化幅度
    const float IMMERSION_THRESHOLD = 8.0f;  // 8 hPa = 约8cm水深
    if (pressure_delta < IMMERSION_THRESHOLD) {
        return false;  // 变化幅度小，可能是大气压
    }
    
    // 3. 检查变化时间跨度
    uint32_t earliest_time = history->timestamps[0];
    uint32_t latest_time = history->timestamps[history->count - 1];
    uint32_t time_span = latest_time - earliest_time;
    
    // 4. 综合判断
    if (pressure_delta >= IMMERSION_THRESHOLD) {
        if (time_span < 300000) {  // 5分钟内
            // 短时间内大幅变化 → 水浸
            return true;
        } else if (time_span < 3600000) {  // 1小时内
            // 中等时间，需要进一步验证
            return (pressure_delta > 12.0f);  // 更高的阈值
        } else {
            // 长时间变化 → 可能是极端天气导致的大气压变化
            return false;
        }
    }
    
    return false;
}
```

### 方法3: 温度关联分析 ??

**原理**: 大气压变化通常伴随温度变化，水浸则温度变化轻微

```c
typedef struct {
    float pressure;
    float temperature;
    uint32_t timestamp;
} sensor_reading_t;

bool detect_immersion_by_temperature_correlation(
    sensor_reading_t *readings, 
    uint8_t count
) {
    if (count < 10) return false;
    
    // 1. 计算压力变化
    float P_delta = readings[count-1].pressure - readings[0].pressure;
    
    // 2. 计算温度变化
    float T_delta = readings[count-1].temperature - readings[0].temperature;
    
    // 3. 分析相关性
    if (fabs(P_delta) > 5.0f) {  // 有显著压力变化
        if (fabs(T_delta) < 0.5f) {
            // 压力变化大，但温度几乎不变 → 可能是水浸
            return true;
        } else {
            // 压力和温度都变化 → 可能是天气变化
            return false;
        }
    }
    
    return false;
}
```

### 方法4: 稳定性分析 ?

**原理**: 水浸后压力会保持稳定，大气压则持续波动

```c
bool detect_immersion_by_stability(pressure_history_t *history) {
    if (history->count < 10) return false;
    
    // 1. 检查是否处于高压状态
    float recent_avg = 0;
    for (int i = history->count - 5; i < history->count; i++) {
        recent_avg += history->pressure_samples[i];
    }
    recent_avg /= 5;
    
    float baseline = 1013.0f;  // 假设的基准
    if (recent_avg - baseline < 8.0f) {
        return false;  // 压力不高，不是水浸
    }
    
    // 2. 计算最近5次采样的方差
    float variance = 0;
    for (int i = history->count - 5; i < history->count; i++) {
        float diff = history->pressure_samples[i] - recent_avg;
        variance += diff * diff;
    }
    variance /= 5;
    
    // 3. 判断稳定性
    const float STABILITY_THRESHOLD = 0.1f;  // 非常稳定
    if (variance < STABILITY_THRESHOLD) {
        // 高压且非常稳定 → 可能是水浸（水深恒定）
        return true;
    } else {
        // 持续波动 → 可能是大气压变化
        return false;
    }
}
```

### 方法5: 连续性验证 ?

**原理**: 水浸需要连续多次确认，避免误判

```c
typedef struct {
    uint8_t immersion_count;      // 连续检测到水浸的次数
    uint8_t normal_count;          // 连续检测到正常的次数
    bool immersion_confirmed;      // 是否确认水浸
} immersion_detector_t;

void update_immersion_detector(
    immersion_detector_t *detector,
    bool suspected_immersion
) {
    const uint8_t CONFIRMATION_THRESHOLD = 3;  // 需要3次连续确认
    
    if (suspected_immersion) {
        detector->immersion_count++;
        detector->normal_count = 0;
        
        if (detector->immersion_count >= CONFIRMATION_THRESHOLD) {
            detector->immersion_confirmed = true;
        }
    } else {
        detector->normal_count++;
        detector->immersion_count = 0;
        
        if (detector->normal_count >= CONFIRMATION_THRESHOLD) {
            detector->immersion_confirmed = false;
        }
    }
}
```

## ? 综合判断策略（推荐）

### 多因子评分系统

```c
typedef struct {
    float rate_score;         // 变化速率得分 (0-1)
    float amplitude_score;    // 变化幅度得分 (0-1)
    float temp_corr_score;    // 温度关联得分 (0-1)
    float stability_score;    // 稳定性得分 (0-1)
    float total_score;        // 总分 (0-4)
} immersion_analysis_t;

immersion_analysis_t analyze_pressure_change(
    pressure_history_t *history,
    float baseline
) {
    immersion_analysis_t result = {0};
    
    // 1. 变化速率评分
    float rate = calculate_pressure_change_rate();
    if (rate > 3.0f) {
        result.rate_score = 1.0f;  // 快速变化，强烈怀疑水浸
    } else if (rate > 1.0f) {
        result.rate_score = 0.5f;  // 中等速度
    } else {
        result.rate_score = 0.0f;  // 缓慢变化，不像水浸
    }
    
    // 2. 变化幅度评分
    float latest = history->pressure_samples[history->count - 1];
    float delta = latest - baseline;
    if (delta > 12.0f) {
        result.amplitude_score = 1.0f;  // 大幅变化
    } else if (delta > 8.0f) {
        result.amplitude_score = 0.7f;  // 中等变化
    } else if (delta > 5.0f) {
        result.amplitude_score = 0.3f;  // 小幅变化
    } else {
        result.amplitude_score = 0.0f;  // 变化很小
    }
    
    // 3. 温度关联评分（逆向：温度变化越小，越像水浸）
    float temp_delta = fabs(latest_temp - initial_temp);
    if (temp_delta < 0.3f) {
        result.temp_corr_score = 1.0f;  // 温度几乎不变，像水浸
    } else if (temp_delta < 1.0f) {
        result.temp_corr_score = 0.5f;
    } else {
        result.temp_corr_score = 0.0f;  // 温度变化大，像天气变化
    }
    
    // 4. 稳定性评分
    float variance = calculate_variance(history);
    if (variance < 0.1f) {
        result.stability_score = 1.0f;  // 非常稳定，像水浸
    } else if (variance < 0.5f) {
        result.stability_score = 0.5f;
    } else {
        result.stability_score = 0.0f;  // 持续波动，像气压变化
    }
    
    // 5. 计算总分
    result.total_score = result.rate_score + 
                        result.amplitude_score + 
                        result.temp_corr_score + 
                        result.stability_score;
    
    return result;
}

// 判断逻辑
bool is_water_immersion_confirmed(immersion_analysis_t *analysis) {
    const float THRESHOLD = 2.5f;  // 总分阈值（满分4分）
    
    if (analysis->total_score >= THRESHOLD) {
        // 多个指标都支持水浸判断
        return true;
    }
    
    // 特殊情况：即使总分不够，但如果变化速率和幅度都很高
    if (analysis->rate_score >= 0.7f && analysis->amplitude_score >= 0.7f) {
        return true;
    }
    
    return false;
}
```

|------|-----------|------|
| ? **速度** | 缓慢 (<0.1 hPa/分钟) | 快速 (>1 hPa/分钟) |

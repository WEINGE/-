  - temp_samples[10]      // 温度数组
  - timestamps[10]        // 时间戳数组

void smart_calibration_add_sample(
    float pressure_hpa, 
    float temperature_c, 
    uint32_t timestamp_ms
) {
    // 添加到环形缓冲区
    buf->pressure_samples[buf->index] = pressure_hpa;
    buf->temp_samples[buf->index] = temperature_c;
    buf->timestamps[buf->index] = timestamp_ms;
    
    buf->index = (buf->index + 1) % 10;  // 环形
    if (buf->count < 10) buf->count++;
}

// 3. 稳定性检查函数
bool smart_calibration_is_environment_stable(void) {
    // 需要至少8次采样
    if (buf->count < 8) {
        return false;  // ← 关键！
    }
    
    // 计算方差
    float pressure_variance = calculate_variance(
        buf->pressure_samples, 
        buf->count
    );
    float temp_variance = calculate_variance(
        buf->temp_samples, 
        buf->count
    );
    
    // 判断是否稳定
    return (pressure_variance < 5.0 && temp_variance < 999.0);
}
